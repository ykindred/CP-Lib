% ============================================================
%  CP-Lib 打印版 —— 导言区
%  引擎：XeLaTeX（中文必需）。用 latexmk -xelatex 编译。

% ---------- 页面：A4 纵向，单栏 ----------
% top 必须容纳 headheight + headsep，否则页眉会被裁掉
\usepackage[a4paper,top=16mm,bottom=15mm,left=16mm,right=16mm,headheight=15pt,headsep=4mm]{geometry}

% ---------- 颜色 ----------
% 单一主色（藏青）+ 中性灰。代码高亮用低饱和色，保证黑白打印仍有层次。
\usepackage{xcolor}
\definecolor{accent}{RGB}{28,62,106}      % 主色：章节号、链接、代码左侧竖条
\definecolor{accent2}{RGB}{146,74,36}     % 次色：仅用于「结论与技巧」小标题
\definecolor{ink}{RGB}{24,24,24}          % 正文黑
\definecolor{muted}{RGB}{108,114,124}     % 页眉页脚、目录页码
\definecolor{hairline}{RGB}{212,220,231}  % 细分隔线
\definecolor{codebg}{RGB}{247,248,251}    % 代码底色（极浅蓝灰）
\definecolor{codekw}{RGB}{18,72,138}      % 关键字
\definecolor{codecmt}{RGB}{72,112,82}     % 注释
\definecolor{codestr}{RGB}{163,50,40}     % 字符串
\definecolor{codedir}{RGB}{118,70,148}    % 预处理指令
\definecolor{linkcolor}{RGB}{28,62,106}

% ---------- 字体 ----------
% ctexart 已自动加载 xeCJK 并选用 Windows 字体（SimSun/SimHei/KaiTi）。
% 代码里的中文注释用黑体：小字号下比宋体笔画更均匀、更清楚。
\setCJKmainfont{SimSun}[AutoFakeBold=2]
\setCJKsansfont{SimHei}[AutoFakeBold=2]
\setCJKmonofont{SimHei}
% 拉丁等宽字体用于代码；Consolas 是 Windows 自带、比赛环境常见
\setmonofont{Consolas}
% 标题用无衬线（Heros + SimHei），与正文的宋体/CM 拉开层次
\setsansfont{TeX Gyre Heros}[Scale=MatchLowercase]

% ---------- 数学 ----------
\usepackage{amsmath}
\usepackage{amssymb}
% 结论里用了 $180\degree$，而 \degree 并非 base LaTeX/amsmath 提供，
% 不定义会直接编译失败。这里补上。
\newcommand{\degree}{\ensuremath{^{\circ}}}

% ---------- 标题层级 ----------
\usepackage{titlesec}
\usepackage{titletoc}
\usepackage{needspace}

% h1 -> \section：主色编号 + 深色标题 + 一条主色细线
\titleformat{\section}[block]
  {\normalfont\sffamily\LARGE\bfseries}
  {\color{accent}\thesection}{0.7em}{\color{ink}}
  [\vspace{2pt}{\color{accent}\titlerule[1.1pt]}]
\titlespacing*{\section}{0pt}{18pt plus 3pt minus 3pt}{8pt}

% h2 -> \subsection：主色编号 + 无衬线标题；不再画线，避免与代码框打架
\titleformat{\subsection}[block]
  {\needspace{5\baselineskip}\normalfont\sffamily\large\bfseries}
  {\color{accent}\thesubsection}{0.55em}{\color{ink}}
\titlespacing*{\subsection}{0pt}{13pt plus 2pt minus 2pt}{4pt}

% 无编号小节目前只有「结论与技巧」：用次色 + 小竖条区分于模板标题
\titleformat{name=\subsection,numberless}[block]
  {\needspace{5\baselineskip}\normalfont\sffamily\large\bfseries\color{accent2}}
  {}{0em}{\textcolor{accent2}{\rule[0.08em]{3.5pt}{1em}}\hspace{0.6em}}

% h3 -> \subsubsection
\titleformat{\subsubsection}
  {\normalfont\sffamily\normalsize\bfseries\color{ink}}
  {\color{accent}\thesubsubsection}{0.5em}{}
\titlespacing*{\subsubsection}{0pt}{7pt}{2pt}

% h4 -> \paragraph（不编号，紧凑）
\titleformat{\paragraph}[runin]
  {\normalfont\sffamily\normalsize\bfseries\color{ink}}
  {}{0em}{}
\titlespacing*{\paragraph}{0pt}{4pt}{0.5em}

% 目录：收 h1/h2（正文里没有 h3），两条目之间留一点呼吸
\setcounter{tocdepth}{2}
\setcounter{secnumdepth}{2}
\contentsmargin{1.7em}
\titlecontents{section}[0em]
  {\addvspace{3pt}\sffamily\bfseries\color{accent}}
  {\thecontentslabel\hspace{0.55em}}{}
  {\color{muted!70}\titlerule*[0.45pc]{.}\hspace{0.55em}\color{accent}\contentspage}
\titlecontents{subsection}[1.5em]
  {}
  {\color{accent}\thecontentslabel\hspace{0.5em}}
  {\color{accent2}\bfseries}
  {\color{muted!70}\titlerule*[0.45pc]{.}\hspace{0.55em}\color{muted}\contentspage}

% ---------- 代码高亮 ----------
\usepackage{listings}
\lstset{
  language=C++,
  % A4 单栏下 8.2pt 足够大；每行最多约 107 列，只有极少数超长行会折行
  basicstyle=\ttfamily\fontsize{8.2}{9.9}\selectfont,
  keywordstyle=\color{codekw}\bfseries,
  commentstyle=\color{codecmt}\itshape,
  stringstyle=\color{codestr},
  directivestyle=\color{codedir},
  directives={include,define,undef,if,ifdef,ifndef,else,elif,endif,pragma},
  % 极浅底色 + 左主色竖条：代码块的边界一眼可见，又不会压过内容
  backgroundcolor=\color{codebg},
  frame=l,
  framerule=1.5pt,
  rulecolor=\color{accent!50},
  framesep=1.5mm,
  xleftmargin=0.4mm,
  xrightmargin=0.4mm,
  % 折行：续行缩进 + 灰色小箭头。prebreak/postbreak 位于 \discretionary 内，
  % 其中不能出现 whatsit（\color 会产生），故把箭头包进 \raisebox 的盒子里。
  breaklines=true,
  breakatwhitespace=false,
  breakindent=1.4em,
  breakautoindent=true,
  prebreak=,
  postbreak=\raisebox{0ex}[0ex][0ex]{\ensuremath{\color{muted}\hookrightarrow}}\space,
  columns=fullflexible,
  keepspaces=true,
  showstringspaces=false,
  tabsize=4,
  extendedchars=true,
  aboveskip=6pt,
  belowskip=6pt,
}
